B. Requirement 1 & 2: Login & OTP (The "Agent" Logic)
We use Redis to store OTPs because it has a built-in "TTL" (Time to Live), meaning the OTP automatically deletes itself after 5 minutes.

Python
# app/services/otp_service.py
import redis
import random
from app.core.config import settings

redis_conn = redis.from_url(settings.REDIS_URL)

async def send_and_store_otp(email: str):
    otp = str(random.randint(100000, 999999))
    # Set OTP with 300 second (5 min) expiry
    redis_conn.setex(f"otp:{email}", 300, otp)
    
    # Mock sending email (In production, use SendGrid/FastAPI-Mail)
    print(f"DEBUG: OTP for {email} is {otp}")
    return {"message": "OTP sent successfully"}

async def verify_otp(email: str, provided_otp: str):
    stored_otp = redis_conn.get(f"otp:{email}")
    if stored_otp and stored_otp.decode() == provided_otp:
        redis_conn.delete(f"otp:{email}") # Delete after successful use
        return True
    return False
C. Requirement 3 & 4: Forgot Password & Authenticated Profile
For profile updates, we use a Dependency Injection called get_current_user to ensure only the owner can edit their data.

Python
# app/api/routes.py
from fastapi import APIRouter, Depends, HTTPException
from app.core.security import get_password_hash, get_current_user

router = APIRouter()

@app.put("/profile/update")
async def update_profile(
    new_data: ProfileUpdateSchema, 
    current_user: User = Depends(get_current_user)
):
    # current_user is automatically verified via JWT
    user_db = db.query(User).filter(User.id == current_user.id).first()
    user_db.username = new_data.username
    db.commit()
    return {"msg": "Profile updated"}

@app.post("/password/forgot")
async def forgot_password(email: str, otp: str, new_password: str):
    if await verify_otp(email, otp):
        user = db.query(User).filter(User.email == email).first()
        user.hashed_password = get_password_hash(new_password)
        db.commit()
        return {"msg": "Password reset successful"}
    raise HTTPException(status_code=400, detail="Invalid OTP")
